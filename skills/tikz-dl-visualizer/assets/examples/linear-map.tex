% Caption: A linear layer moves the whole input space at once. Grid lines stay
% straight, parallel and evenly spaced; the basis vectors e1, e2 land on the
% columns of W. Everything the layer does is decided by where those two land.
% Archetype: before/after panels in input space (TikZ cm= transform)
\documentclass[tikz,border=8pt]{standalone}
\usepackage[dark]{dlviz}
\begin{document}
\begin{tikzpicture}
  % W = [[1.2, -0.4], [0.6, 0.9]]  -> cm={a, b, c, d, (shift)} takes
  % (x,y) to (a x + c y, b x + d y): first column (a,b), second (c,d).
  % cm needs literal numbers; a macro holding them will not parse.

  % ---- before ----------------------------------------------------------------
  \begin{scope}
    \clip (-2.2,-2.2) rectangle (2.2,2.2);
    \draw[dlgrid] (-3,-3) grid (3,3);
  \end{scope}
  \draw[->, dlfaint] (-2.2,0) -- (2.3,0);
  \draw[->, dlfaint] (0,-2.2) -- (0,2.3);
  \fill[colorpink, opacity=.25] (0,0) rectangle (1,1);
  \draw[->, very thick, colororange] (0,0) -- (1,0) node[below right] {\(\ve_1\)};
  \draw[->, very thick, colorblue]   (0,0) -- (0,1) node[above left]  {\(\ve_2\)};
  \node[dim, anchor=north] at (0,-2.3) {input space \(\pink{\vx}\in\R^2\)};

  % ---- arrow between panels ----------------------------------------------------
  \draw[flow] (2.6,0) -- (3.6,0) node[midway, above] {\(\mW\)};

  % ---- after -----------------------------------------------------------------
  \begin{scope}[xshift=6.2cm]
    \begin{scope}
      \clip (-2.2,-2.2) rectangle (2.2,2.2);
      \draw[dlgrid] (-3,-3) grid (3,3);
      \begin{scope}[cm={1.2,0.6,-0.4,0.9,(0,0)}]
        \draw[colorgreen!45!dlbg] (-4,-4) grid (4,4);
        \fill[colorgreen, opacity=.25] (0,0) rectangle (1,1);
      \end{scope}
    \end{scope}
    \draw[->, dlfaint] (-2.2,0) -- (2.3,0);
    \draw[->, dlfaint] (0,-2.2) -- (0,2.3);
    \draw[->, very thick, colororange] (0,0) -- (1.2,0.6)
      node[right] {\(\mW\ve_1\)};
    \draw[->, very thick, colorblue]   (0,0) -- (-0.4,0.9)
      node[above left] {\(\mW\ve_2\)};
    \node[dim, anchor=north] at (0,-2.3)
      {\(\green{\vh} = \mW\pink{\vx}\); areas scale by \(|\det\mW| = 1.32\)};
  \end{scope}

  % ---- the matrix, with columns coloured like the vectors they become ----------
  \node[anchor=north] at (3.1,-3.0) {%
    \(\mW = \begin{bmatrix} \orange{1.2} & \blue{-0.4} \\
                            \orange{0.6} & \blue{\phantom{-}0.9} \end{bmatrix}
      \qquad \text{column } j \text{ of } \mW \text{ is where } \ve_j \text{ lands}\)};
\end{tikzpicture}
\end{document}
