% Caption: On an ill-conditioned quadratic loss (curvature 10x larger along w2),
% plain gradient descent bounces across the narrow valley while creeping along
% it. Momentum averages successive gradients: the alternating w2 components
% cancel and the consistent w1 component builds up.
% Archetype: geometry in parameter space (contours + trajectories)
% Trajectories computed in Python (L = (w1^2 + 10 w2^2)/2, 12 steps each).
\documentclass[tikz,border=8pt]{standalone}
\usepackage[dark]{dlviz}
\begin{document}
\begin{tikzpicture}[scale=1.15]
  % level sets L = c are ellipses with radii sqrt(2c) and sqrt(2c/10)
  \foreach \c in {0.5,2,4.5,8,12.5} {
    \pgfmathsetmacro\rx{sqrt(2*\c)} \pgfmathsetmacro\ry{sqrt(2*\c/10)}
    \draw[dlfaint, thin] (0,0) ellipse ({\rx} and {\ry});
  }
  \fill[colorred] (0,0) circle (1.6pt) node[below right=-1pt, font=\scriptsize,
    text=colorred] {\(\vw^\star\)};
  \draw[->, dlgrid] (-5.3,0) -- (5.3,0) node[right, text=dlmuted] {\(w_1\)};
  \draw[->, dlgrid] (0,-1.9) -- (0,1.9) node[above, text=dlmuted] {\(w_2\)};

  % plain GD, lr = 0.18
  \draw[colororange, thick, -{Latex[length=1.6mm]}]
    plot coordinates {(-4.00,1.50) (-3.28,-1.20) (-2.69,0.96) (-2.21,-0.77)
      (-1.81,0.61) (-1.48,-0.49) (-1.22,0.39) (-1.00,-0.31) (-0.82,0.25)
      (-0.67,-0.20) (-0.55,0.16) (-0.45,-0.13) (-0.37,0.10)};
  \foreach \p in {(-4.00,1.50),(-3.28,-1.20),(-2.69,0.96),(-2.21,-0.77),
      (-1.81,0.61),(-1.48,-0.49),(-1.22,0.39),(-1.00,-0.31)}
    \fill[colororange] \p circle (1.1pt);

  % momentum, lr = 0.06, beta = 0.6
  \draw[coloraqua, thick, smooth, -{Latex[length=1.6mm]}]
    plot coordinates {(-4.00,1.50) (-3.76,0.60) (-3.39,-0.30) (-2.97,-0.66)
      (-2.53,-0.48) (-2.12,-0.08) (-1.75,0.20) (-1.42,0.25) (-1.13,0.13)
      (-0.90,-0.02) (-0.70,-0.10) (-0.54,-0.09) (-0.41,-0.03)};

  % legend as direct labels, not a box
  \node[annot, knockout, text=colororange, anchor=south west] at (-2.55,1.05)
    {gradient descent: steep \(w_2\) direction\\dominates, so it zig-zags};
  \node[annot, knockout, text=coloraqua, anchor=north] at (-2.2,-1.3)
    {momentum: alternating pushes cancel,\\steady \(w_1\) pushes accumulate};
  \draw[leader, coloraqua] (-2.2,-1.25) -- (-2.6,-0.6);

  \node[anchor=north] at (0,-2.25) {%
    \(\orange{\vw_{t+1} = \vw_t - \eta\nabla\cL(\vw_t)}\)\qquad
    \(\aqua{\vv_{t+1} = \beta\vv_t + \nabla\cL(\vw_t),\;
      \vw_{t+1} = \vw_t - \eta\vv_{t+1}}\)};
\end{tikzpicture}
\end{document}
