% Caption: Scaled dot-product attention with a temperature. Hover a token (a row
% of Q, A or AV) to follow its query; slide tau to watch softmax sharpen into
% argmax or flatten into a plain average.
% Scaffold: Q, K^T, V with their (fixed) values, empty A and AV cells, labels,
% equation. Live (attention.js): A heatmap + weights, AV values, tau readout,
% row highlights and weight curves to V.
\documentclass[tikz,border=8pt]{standalone}
\usepackage[dark]{dlviz}
\usepackage{dllive}
\begin{document}
\begin{tikzpicture}[x=6mm, y=6mm]
  \useasboundingbox (-7.2,-8.4) rectangle (11.8,7.4);
  \def\tokens{the,cat,sat,.}
  % Q, K, V from attention.js (one decimal, so the printed values are exact)
  \def\Q{{2.0,.8,0},{.4,2.2,.3},{0,2.8,.4},{.5,.5,.6}}
  \def\KT{{1.2,.3,0,.4},{.1,1.4,.4,.3},{.2,.1,1.3,.2}}
  \def\V{{.9,.1,-.2},{-.3,1.0,.4},{.2,-.5,.8},{0,.2,.1}}
  \tikzset{c/.style={cell=#1, minimum size=6mm}}

  % ---- A (4x4): outlines only; fill + weights are live ----------------------
  \foreach \i in {1,...,4} \foreach \j in {1,...,4} {
    \node[c=coloraqua, fill=none, export] (A-\i-\j) at (\j-.5, -\i+.5) {};
    \node[live=a-\i-\j, anchor=mid west, font=\scriptsize] at (A-\i-\j.center) {\(\dlnum\)};
  }
  \foreach \i in {1,...,4}
    \node[fit=(A-\i-1)(A-\i-4), inner sep=0, export] (Arow-\i) {};
  \node[dim, below=1mm of A-4-2.south east] {\(\aqua{\mA} \in \R^{n\times n}\)};

  % ---- Q (4x3) left of A ------------------------------------------------------
  \foreach \row [count=\i] in \Q {
    \foreach \v [count=\j] in \row
      \node[c=colorblue, font=\scriptsize] (Q-\i-\j) at (-5.5+\j, -\i+.5) {\(\v\)};
  }
  \foreach \t [count=\i] in \tokens {
    \node[left=1mm of Q-\i-1, font=\small] (tok-\i) {\t};
    \node[fit=(tok-\i)(Q-\i-1)(Q-\i-3), inner sep=0, export] (Qrow-\i) {};
  }
  \node[dim, below=1mm of Q-4-2] {\(\blue{\mQ} \in \R^{n\times d_k}\)};

  % ---- K^T (3x4) above A -------------------------------------------------------
  \foreach \row [count=\i] in \KT
    \foreach \v [count=\j] in \row
      \node[c=colororange, font=\scriptsize] (K-\i-\j) at (\j-.5, 4.5-\i) {\(\v\)};
  \foreach \t [count=\j] in \tokens \node[above=1mm of K-1-\j, font=\small] {\t};
  \node[dim, left=1mm of K-2-1] {\(\orange{\mK}^{\top}\)};

  % ---- V (4x3) above the output; output AV (4x3) right of A -------------------
  \foreach \row [count=\i] in \V {
    \foreach \v [count=\j] in \row
      \node[c=colorgreen, font=\scriptsize] (V-\i-\j) at (6.5+\j, 6.5-\i) {\(\v\)};
  }
  \foreach \t [count=\i] in \tokens {
    \node[right=1mm of V-\i-3, font=\small] {\t};
    \node[fit=(V-\i-1)(V-\i-3), inner sep=0, export] (Vrow-\i) {};
  }
  \node[dim, above=1mm of V-1-2] {\(\green{\mV}\)};
  \foreach \i in {1,...,4} {
    \foreach \j in {1,2,3} {
      \node[c=colorgreen, export] (O-\i-\j) at (6.5+\j, -\i+.5) {};
      \node[live=o-\i-\j, anchor=mid west, font=\scriptsize] at (O-\i-\j.center) {\(\dlnum\)};
    }
    \node[fit=(O-\i-1)(O-\i-3), inner sep=0, export] (Orow-\i) {};
  }
  \node[dim, below=1mm of O-4-2] {\(\mA\green{\mV} \in \R^{n\times d_v}\)};

  % ---- equation with temperature, and the live tau readout --------------------
  \node[anchor=north] at (2,-5.6) {%
    \(\mathrm{Attn}(\blue{\mQ},\orange{\mK},\green{\mV}) =
      \underbrace{\softmax\!\Bigl(\tfrac{\blue{\mQ}\orange{\mK}^{\top}}{\tau\sqrt{d_k}}\Bigr)}_{\aqua{\mA}}\,
      \green{\mV}\)};
  \node[anchor=base west, inner sep=0, font=\small, text=dlmuted] (tau) at (-6.9,6.4) {\(\tau = {}\)};
  \node[live=tau, anchor=base west, inner sep=0, font=\small, text=dlmuted] at (tau.base east) {\(\dlnum\)};
\end{tikzpicture}
\end{document}
